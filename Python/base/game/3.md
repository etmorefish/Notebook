这里有一个简单的贪吃蛇游戏，使用基本的 Python 数据结构（列表、元组、字典、集合）来实现。以下是一个基于文本的贪吃蛇游戏示例，不使用任何特殊库。

### 基于文本的贪吃蛇游戏

```python
import random
import os
import time

# 游戏设置
WIDTH = 20
HEIGHT = 10
SNAKE_CHAR = "🤖"
FOOD_CHAR = "🍎"
EMPTY_CHAR = "✉️"

# 初始化游戏状态
game_state = {
    "score": 0,
    "snake": [(HEIGHT // 2, WIDTH // 2)],  # 初始化蛇
    "food": None,
    "direction": "d",  # 初始方向
    "food_positions": set(),  # 存储已生成的食物位置
}


def generate_food():
    """生成食物"""
    while True:
        food = (random.randint(0, HEIGHT - 1), random.randint(0, WIDTH - 1))
        if food not in game_state["snake"] and food not in game_state["food_positions"]:
            game_state["food_positions"].add(food)  # 将食物位置添加到集合
            return food


def create_game_area():
    """创建游戏区域"""
    return [[EMPTY_CHAR for _ in range(WIDTH)] for _ in range(HEIGHT)]


def print_game_area():
    """打印游戏区域"""
    os.system("cls" if os.name == "nt" else "clear")  # 清屏
    print("贪吃蛇游戏")
    print(f"得分: {game_state['score']}")
    for y in range(HEIGHT):
        for x in range(WIDTH):
            if (y, x) == game_state["food"]:
                print(FOOD_CHAR, end=" ")
            elif (y, x) in game_state["snake"]:
                print(SNAKE_CHAR, end=" ")
            else:
                print(EMPTY_CHAR, end=" ")
        print()
    print("使用 WASD 控制蛇的移动，按 Q 退出。")


def move_snake():
    """移动蛇"""
    head = game_state["snake"][0]
    direction = game_state["direction"]

    match direction:
        case "w":
            new_head = (head[0] - 1, head[1])
        case "s":
            new_head = (head[0] + 1, head[1])
        case "a":
            new_head = (head[0], head[1] - 1)
        case "d":
            new_head = (head[0], head[1] + 1)
        case _:
            return None  # 无效的移动

    # 检查碰撞
    if (
        new_head in game_state["snake"]
        or new_head[0] < 0
        or new_head[0] >= HEIGHT
        or new_head[1] < 0
        or new_head[1] >= WIDTH
    ):
        return None

    game_state["snake"].insert(0, new_head)  # 在蛇头插入新位置
    return new_head


def main():
    """游戏主函数"""
    game_state["food"] = generate_food()  # 生成食物

    while True:
        print_game_area()

        # 获取用户输入
        move = input("移动 (WASD): ").lower()
        if move == "q":
            print("退出游戏")
            break
        elif move in ["w", "s", "a", "d"]:
            game_state["direction"] = move

        new_head = move_snake()

        if new_head is None:
            print("游戏结束！")
            break

        # 检查是否吃到食物
        if new_head == game_state["food"]:
            game_state["food"] = generate_food()  # 重新生成食物
            game_state["score"] += 1  # 增加得分
        else:
            game_state["snake"].pop()  # 移除蛇尾

        time.sleep(0.1)  # 控制游戏速度


if __name__ == "__main__":
    main()

```

### 代码简介

1. **数据结构**：
   - **列表**：用于存储蛇的位置，蛇的每个部分用元组表示坐标。
   - **元组**：用于表示蛇的每个部分的位置（行和列）。
   - **集合**：可用于存储食物的位置、检查食物生成。
   - **字典**：使用字典存储游戏状态，将 score、snake、food 和 direction 存储在 game_state 字典中，以便于管理和访问。

   - **生成食物**：随机生成食物的位置，确保不与蛇的位置重叠。
